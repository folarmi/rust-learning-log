31/// Returns the nth Fibonacci number, counting from F(0) = 0.
32///
33/// F(0) = 0, F(1) = 1, and every number after is the sum of the two before it:
34/// F(n) = F(n-1) + F(n-2)
35///
36/// How it works: we only ever keep the last two numbers, `prev` and `curr`,
37/// and slide them forward one step per loop. After n steps, `prev` holds F(n).
38///
39/// | step  | prev | curr |
40/// |-------|------|------|
41/// | start | 0    | 1    |
42/// | 1     | 1    | 1    |
43/// | 2     | 1    | 2    |
44/// | 3     | 2    | 3    |
45///
46/// Limit: n must be 92 or less. Each step also calculates the NEXT number,
47/// so for n = 93 the loop tries to compute F(94), which is too big for u64.
48fn fibonacci(n: u64) -> u64 {
49    let mut prev = 0;
50    let mut curr = 1;
51
52    for _ in 0..n {
53        // Right side is calculated FIRST, using the old values:
54        //   new prev = old curr
55        //   new curr = old prev + old curr  (the sum, i.e. the "next" number)
56        // Then both are assigned at once, so nothing is overwritten too early.
57        // Same as JS destructuring: [prev, curr] = [curr, prev + curr];
58        (prev, curr) = (curr, prev + curr);
59    }
60
61    prev
62}
